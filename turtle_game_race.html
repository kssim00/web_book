STEP 1

01게임판 그리기

화면 크기를 정하고, 제목을 화면 위에 적어요.

import turtle as t
import random as r

t.screensize(600, 400, "lavender")  # 화면 크기와 배경색
t.up()
t.ht()                               # 거북이 모양 숨기기
t.goto(-150, 220)
t.write("터틀레이스게임", "center", font=("Arial", 28))  # 제목 쓰기
t.ht()는 거북이 모양을 숨겨요. t.write()로 화면에 글자를 적을 수 있어요.

STEP 4

04거북이 여러 마리 준비하기

반복문으로 거북이 객체를 여러 개 만들고, 색을 랜덤으로 정한 뒤 레인 번호를 옆에 적어요.

turtles = []
sety = 180
for i in range(7):
    a = t.Turtle()
    a.ht()
    r1 = r.random()
    g1 = r.random()
    b1 = r.random()
    a.color(r1, g1, b1)  # 랜덤 색
    a.shape("turtle")
    a.up()
    a.st()
    a.goto(-350, sety)
    t.up()
    t.goto(-370, sety)
    t.write(i)            # 레인 번호 쓰기
    sety -= 80
    turtles.append(a)
만든 거북이는 turtles 리스트에 하나씩 담아둬요. 이 리스트로 나중에 모든 거북이를 한 번에 움직여요.

06경주 진행시키기

거북이들이 랜덤 거리만큼 전진해요. 한 마리라도 도착선을 넘으면 경주가 끝나요.

def racer():
    game_over = True
    while game_over:
        for tu in turtles:
            rspeed = r.randint(1, 20)  # 랜덤 속도
            tu.fd(rspeed)
            if tu.xcor() > 280:         # 도착선 통과 확인
                game_over = False

racer()

STEP 7

07승자 발표

x좌표가 가장 큰 거북이가 1등이에요. 배팅한 번호와 비교해서 성공·실패를 알려줘요.

max_xcor = 0
winner = 0
for i in range(len(turtles)):
    if turtles[i].xcor() > max_xcor:
        max_xcor = turtles[i].xcor()
        winner = i  # 가장 멀리 간 거북이

if user_choice == winner:
    t.up()
    t.goto(0, 0)
    t.color("black")
    t.write("성공!", False, "center", ("bold", 20))
else:
    t.write(f"실패! {winner}번 거북이 승!!")

t.mainloop()
위 STEP 6 미리보기에서 도착한 거북이 번호가 winner예요. 배팅 번호(user_choice)와 같으면 성공, 다르면 실패를 알려줘요.

08클래스로 정리하기

거북이 하나마다 색 지정 → 모양 지정 → 위치 이동 코드가 반복돼요. 이걸 클래스로 묶으면 거북이 하나당 한 줄로 만들 수 있어요.

class Racer:
    def __init__(self, y):
        self.turtle = t.Turtle()
        self.turtle.shape("turtle")
        r1 = r.random()
        g1 = r.random()
        b1 = r.random()
        self.turtle.color(r1, g1, b1)  # 랜덤 색
        self.turtle.up()
        self.turtle.goto(-330, y)

    def race(self):
        rspeed = r.randint(1, 20)
        self.turtle.fd(rspeed)

    def xcor(self):
        return self.turtle.xcor()
__init__ 안에 색·모양·위치 지정을 다 넣었기 때문에, 거북이를 만들 때는 Racer(y좌표) 한 줄이면 충분해요.
tt = [Racer(180), Racer(100), Racer(20), Racer(-60), Racer(-140), Racer(-220)]

game_over = False
while not game_over:
    for i in range(6):
        tt[i].race()
        if tt[i].xcor() >= 300:  # 도착선 통과
            game_over = True
            break

t.mainloop()
동작 결과는 STEP 6~7과 완전히 같아요. 코드만 클래스로 깔끔하게 정리된 거예요.
